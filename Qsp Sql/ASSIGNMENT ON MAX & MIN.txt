ASSIGNMENT ON MAX & MIN

1)  SELECT ENAME FROM EMP
 WHERE SAL = (SELECT MAX(SAL) FROM EMP);

2)  SELECT ENAME FROM EMP
 WHERE SAL = (SELECT MIN(SAL) FROM EMP);

3)  SELECT ENAME,HIREDATE FROM EMP
WHERE HIREDATE = (SELECT MIN(HIREDATE) FROM EMP);

4)  SELECT ENAME,HIREDATE FROM EMP
WHERE HIREDATE = (SELECT MAX(HIREDATE) FROM EMP);

5)  SELECT ENAME,COMM FROM EMP
 WHERE COMM = (SELECT MIN(COMM) FROM EMP);

6)  SELECT ENAME,SAL,COMM FROM EMP
WHERE COMM = (SELECT MAX(COMM) FROM EMP);

7)  SELECT * FROM EMP
WHERE EMPNO = (SELECT MAX(EMPNO) FROM EMP);

8)  SELECT * FROM EMP
WHERE HIREDATE = (SELECT MIN(HIREDATE) FROM EMP);

9)  SELECT * FROM EMP
WHERE SAL*12 = (SELECT MIN(SAL*12) FROM EMP);

10) SELECT ENAME,SAL*12 AS ANNUALSAL FROM EMP
WHERE SAL*12 > ALL (SELECT SAL*12 FROM EMP
WHERE JOB = 'SALESMAN');
